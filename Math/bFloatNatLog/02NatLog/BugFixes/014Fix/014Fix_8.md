```go
// naturalLogAGM.go
package naturalLogCalcs

import "math/big"

type naturalLogAGM struct{}

// lnAGMDirect
//
// Computes ln(x) using the AGM-based algorithm only (no Taylor fallback).
// Preconditions:
//   - x > 0
//   - precBits > 0
//
// On success, returns ln(x) as *big.Float with precision = precBits.
func (nlAGM *naturalLogAGM) lnAGMDirect(
	x *big.Float,
	precBits uint) (*big.Float, error) {

	// Nanobot
	nano := new(naturalLogAGMNanobot)

	normData, err := nano.reduceByPowersOfTwo(x, precBits)
	if err != nil {
		return nil, err
	}

	modulus, err := nano.computeModulusFromXReduced(normData, precBits)
	if err != nil {
		return nil, err
	}

	agmVal, err := nano.agmKernel(modulus, precBits)
	if err != nil {
		return nil, err
	}

	lnX, err := nano.computeLnFromAGM(normData, modulus, agmVal, precBits)
	if err != nil {
		return nil, err
	}

	return lnX, nil
}
```

### Page 4 — AGM core iteration and tests (high level)

#### AGM core iteration (Macrobot)

Mathematically, the AGM of $a_0$ and $b_0$ is defined by:

$$ \begin{aligned} a_{n+1} &= \frac{a_n + b_n}{2}, \ b_{n+1} &= \sqrt{a_n b_n}, \end{aligned} $$

and $\operatorname{AGM}(a_0, b_0) = \lim_{n\to\infty} a_n = \lim_{n\to\infty} b_n$.

Skeleton:

```go
func (macro *naturalLogAGMMacrobot) agmCore(
	a0, b0 *big.Float,
	precBits uint) (*big.Float, error) {

	// Working precision
	workingPrec := precBits + 16

	newF := func() *big.Float {
		return new(big.Float).
			SetMode(big.AwayFromZero).
			SetPrec(workingPrec)
	}

	a := newF().Set(a0)
	b := newF().Set(b0)

	sqrtCalc := new(BigFloatMath)

	for {
		// aNext = (a + b)/2
		aNext := newF()
		aNext.Add(a, b)
		aNext.Quo(aNext, newF().SetFloat64(2.0))

		// bNext = sqrt(a * b)
		prod := newF().Mul(a, b)

		bNext, err := sqrtCalc.SqrtBigFloat(prod, 0, workingPrec)
		if err != nil {
			return nil, err
		}

		// Convergence check
		if aNext.Cmp(bNext) == 0 {
			a = aNext
			break
		}

		a, b = aNext, bNext
	}

	out := new(big.Float).
		SetMode(big.AwayFromZero).
		SetPrec(precBits)

	out.Set(a)
	return out, nil
}
```

```go
func Test_lnAGM_vs_lnTaylor_Grid_01(t *testing.T) {
	ePrefix := "Test_lnAGM_vs_lnTaylor_Grid_01"

	bh := new(BigFloatHelper)
	precBits := bh.ComputeBigFloatPrecisionBits(200, 4)

	disp := new(naturalLogDispatcher)
	taylor := new(naturalLogTaylor)
	agm := new(naturalLogAGM)

	testXs := []string{
		"0.125", "0.25", "0.5", "0.75",
		"1.0", "1.5", "2.0", "3.0", "5.0", "10.0",
	}

	for _, xStr := range testXs {
		x, ok := new(big.Float).
			SetMode(big.AwayFromZero).
			SetPrec(precBits).
			SetString(xStr)
		if !ok {
			t.Fatalf("%v\nSetString failed for x=%v\n", ePrefix, xStr)
		}

		lnT, err := taylor.lnTaylorDirect(x, precBits)
		if err != nil {
			t.Fatalf("%v\nlnTaylorDirect error: %v\n", ePrefix, err)
		}

		lnA, err := agm.lnAGMDirect(x, precBits)
		if err != nil {
			t.Fatalf("%v\nlnAGMDirect error: %v\n", ePrefix, err)
		}

		decDigits := bh.ComputeBigFloatDecimalDigits(precBits, 0)
		compareDigits := int(decDigits / 2)

		tStr := lnT.Text('f', compareDigits)
		aStr := lnA.Text('f', compareDigits)

		if tStr != aStr {
			t.Errorf("%v\nMismatch lnTaylor vs lnAGM for x=%v\nTaylor: %v\nAGM:    %v\n",
				ePrefix, xStr, tStr, aStr)
		}
	}
}
```
