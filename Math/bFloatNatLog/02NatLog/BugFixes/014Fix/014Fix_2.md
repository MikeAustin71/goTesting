// naturalLogAGMMacrobot.go
package naturalLogCalcs

import (
	"math/big"
	ePref "github.com/MikeAustin71/errpref"
)

type naturalLogAGMMacrobot struct{}

// lnAGMKernel
//
// High‑level AGM pipeline:
//   1) Normalize x into a suitable range for AGM.
//   2) Run the true AGM iteration to compute M(a0, b0).
//   3) Apply the AGM‑based ln(x) formula.
//   4) Re‑apply any scaling (e.g., powers of 2, ln(2), etc.).
func (m *naturalLogAGMMacrobot) lnAGMKernel(
	x *big.Float,
	precBits uint,
	errPrefDto *ePref.ErrPrefixDto) (*big.Float, error) {

	ePrefix, err := ePref.ErrPrefixDto{}.NewFromErrPrefDto(
		errPrefDto,
		"naturalLogAGMMacrobot.lnAGMKernel()",
		"")
	if err != nil {
		return new(big.Float), err
	}

	// 1) Normalize input (range reduction, sign checks, etc.)
	norm := new(naturalLogAGMNanobot)

	normX, scaleFactor, err := norm.normalizeForAGM(
		x,
		precBits,
		ePrefix)
	if err != nil {
		return new(big.Float), err
	}

	// 2) Run AGM iteration: M(a0, b0)
	agmIter := new(naturalLogAGMMolecule)

	a0, b0 := norm.initialAGMPair(normX, precBits)

	M, err := agmIter.trueAGM(
		a0,
		b0,
		precBits,
		ePrefix)
	if err != nil {
		return new(big.Float), err
	}

	// 3) Apply AGM ln formula (pure AGM, no Taylor)
	lnCore, err := norm.lnFromAGM(
		normX,
		M,
		precBits,
		ePrefix)
	if err != nil {
		return new(big.Float), err
	}

	// 4) Re‑apply scaling (e.g., ln(x) = ln(normX) + scaleFactor * ln(2))
	lnFinal, err := norm.applyAGMScaling(
		lnCore,
		scaleFactor,
		precBits,
		ePrefix)
	if err != nil {
		return new(big.Float), err
	}

	return lnFinal, nil
}

# **Do We Need “Lowest Common Denominator” or Fraction Simplification in the AGM Algorithm?**

Short answer: **No. Absolutely not.**
 Longer answer: **The AGM algorithm never requires rational‑number normalization, fraction simplification, or lowest‑common‑denominator operations.**

Let me explain why with precision.

------

# **1. AGM Uses Only Real‑Valued Operations**

The classical AGM iteration uses only:

- Addition
- Division by 2
- Multiplication
- Square root

The iteration is:

$$ a_{n+1} = \frac{a_n + b_n}{2} $$

$$ b_{n+1} = \sqrt{a_n b_n} $$

These operations are **purely real‑valued**.
 There is **no rational arithmetic**, no numerator/denominator tracking, and no simplification step.

Your implementation uses `big.Float`, which is a **floating‑point** representation, not a rational representation. Therefore:

- There are **no fractions** to simplify.
- There is **no denominator** to reduce.
- There is **no need for GCD, LCM, or rational normalization**.

These transformations involve:

- Subtraction

- Square root

Again, **all real operations**, no rational simplification.
